**Goal:** Tăng độ chính xác bước lọc nhiễu transcript bằng confidence metrics, phát hiện lặp/lời nhạc tốt hơn, và LLM mạnh hơn — theo thiên hướng giữ-dialogue thận trọng.

**Architecture:** `transcribe_backends.py` luồn thêm metric tùy chọn (`no_speech_prob`/`avg_logprob`/`compression_ratio`) vào entries (Groq đầy đủ, local best-effort). `clean_transcript.py` thêm `confidence_is_noise`, `is_repetitive_text`, `find_global_music_keys`, `_is_near_dup`, tích hợp vào `tag_entries_heuristic`, và nâng `build_llm_prompt`/`llm_clean_batch`.

**Tech Stack:** Python 3.10+, stdlib (`re`, `difflib`, `collections`), groq SDK, pytest. Không thêm dependency, không train model.

## Global Constraints

- Python 3.10+ (`str | None`, `list[dict]`).
- Schema chỉ THÊM field tùy chọn (`no_speech_prob`, `avg_logprob`, `compression_ratio`) — KHÔNG đổi/bỏ field cũ; `split_audio.py` không đụng.

- Tương thích ngược: entry không có metric chạy như cũ; LLM model là tham số có default mới; mọi test cũ phải vẫn xanh.
- Comment/code/commit message tiếng Việt-không-dấu hoặc tiếng Anh; KHÔNG đưa xưng hô anh/em vào code.

NO_SPEECH_MAX = 0.6
COMPRESSION_MAX = 2.4
LOGPROB_MIN = -1.0
LOGPROB_NOSPEECH_COMBO = 0.4


def confidence_is_noise(entry: dict) -> bool:
    ns = entry.get("no_speech_prob")
    cr = entry.get("compression_ratio")
    lp = entry.get("avg_logprob")
    if ns is not None and ns >= NO_SPEECH_MAX:
        return True
    if cr is not None and cr >= COMPRESSION_MAX:
        return True
    if lp is not None and ns is not None and lp <= LOGPROB_MIN and ns >= LOGPROB_NOSPEECH_COMBO:
        return True
    return False
```

REPEAT_MIN_TOKENS = 4
REPEAT_UNIQUE_RATIO = 0.5


def is_repetitive_text(text: str) -> bool:
    tokens = re.findall(r"\w+", text.lower())
    if len(tokens) < REPEAT_MIN_TOKENS:
        return False
    return (len(set(tokens)) / len(tokens)) <= REPEAT_UNIQUE_RATIO
```

SIM_THRESHOLD = 0.85
GLOBAL_FREQ_MIN = 4
GLOBAL_PHRASE_MAX_WORDS = 12


def _is_near_dup(a: str, b: str) -> bool:
    return difflib.SequenceMatcher(None, a, b).ratio() >= SIM_THRESHOLD


def find_global_music_keys(entries: list[dict]) -> set[str]:
    counts: Counter = Counter()
    for e in entries:
        key = _norm_key(e.get("text_raw", e.get("text", "")))
        if key and 1 <= len(key.split()) <= GLOBAL_PHRASE_MAX_WORDS:
            counts[key] += 1
    return {k for k, c in counts.items() if c >= GLOBAL_FREQ_MIN}
```

def tag_entries_heuristic(entries: list[dict], repeat_threshold: int = 3) -> list[dict]:
    out: list[dict] = []
    for e in entries:
        raw = e.get("text", "")
        new = dict(e)
        new["text_raw"] = raw
        new["text"] = raw
        if is_sound_tag(raw):
            new["type"] = TYPE_SOUND
        elif confidence_is_noise(new):
            new["type"] = TYPE_NOISE
        elif is_hallucination(raw):
            new["type"] = TYPE_NOISE
        elif is_repetitive_text(raw):
            new["type"] = TYPE_MUSIC
        else:
            new["type"] = TYPE_DIALOGUE
        out.append(new)

    # tan suat toan bai -> music (chi dialogue)
    music_keys = find_global_music_keys(out)
    if music_keys:
        for o in out:
            if o["type"] == TYPE_DIALOGUE and _norm_key(o["text_raw"]) in music_keys:
                o["type"] = TYPE_MUSIC

    # near-duplicate lien tiep >= threshold -> music (chi dialogue)
    i = 0
    n = len(out)
    while i < n:
        j = i
        ki = _norm_key(out[i]["text_raw"])
        while j + 1 < n and ki and _is_near_dup(ki, _norm_key(out[j + 1]["text_raw"])):
            j += 1
        if (j - i + 1) >= repeat_threshold:
            for k in range(i, j + 1):
                if out[k]["type"] == TYPE_DIALOGUE:
                    out[k]["type"] = TYPE_MUSIC
        i = j + 1
    return out
```
